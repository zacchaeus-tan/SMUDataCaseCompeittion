// A field shows (and is required) when its rule is "Always", or when its condition is met.
VisibleFields = Filter(
    TypeRules,

);
MissingFields = Filter(VisibleFields, IsBlank(Trim(LookUp(FieldValues, F = FieldName).V)));

// ---- Filled in, but plausible? Placeholders (N/A, TBC, -) never pass; formats come from the field's Check
// (RequiredFields, editable on the Rules screen), else the default below. k = the value's letters and digits only.
DefaultChecks = Table(
    {F: "CustomerName", C: "Text"},
    {F: "RegistrationNo", C: "RegNo"},
    {F: "BillingAddress", C: "Text"},
    {F: "ContactEmail", C: "Email"},
    {F: "ItemDescription", C: "Text"},
    {F: "ManufacturerPartNo", C: "PartNo"},
    {F: "HSCode", C: "HSCode"},
    {F: "RegulatoryLicenceNo", C: "LicenceNo"},
    {F: "ShelfLifeMonths", C: "Months"},
    {F: "VendorName", C: "Text"},
    {F: "TaxID", C: "TaxID"},
    {F: "ISO13485CertNo", C: "LicenceNo"},
    {F: "BankName", C: "Text"},
    {F: "BankAccount", C: "BankAccount"},
    {F: "CostCenterName", C: "Text"},
    {F: "Owner", C: "Text"},
    {F: "BudgetYear", C: "Year"}
);
FreeTextFields = ["CustomerName", "RegistrationNo", "BillingAddress", "ContactEmail", "ItemDescription", "ManufacturerPartNo", "HSCode", "RegulatoryLicenceNo", "VendorName", "Country", "TaxID", "ISO13485CertNo", "BankName", "BankAccount", "CostCenterName", "Owner"];
NumberFields = ["CreditLimit", "ShelfLifeMonths", "BudgetYear"];
Placeholders = ["na", "nil", "none", "null", "tbc", "tba", "tbd", "x", "xx", "xxx", "xxxx", "unknown", "test", "dummy", "pending", "later", "asap", "idk", "same", "sameasabove", "asabove", "abc", "asdf", "qwerty", "123", "1234", "12345", "0", "00", "000"];
FieldChecks = ForAll(
    Filter(VisibleFields, FieldName in FreeTextFields || FieldName in NumberFields) As r,
    With(
        {v: Trim(LookUp(FieldValues, F = r.FieldName).V),
         c: Coalesce(r.Check, LookUp(DefaultChecks, F = r.FieldName).C, "None")},
        With(
            {k: Lower(Concat(Filter(ForAll(Sequence(Len(v)), Mid(v, Value, 1)) As ch, IsMatch(ch.Value, "[A-Za-z0-9]")), Value))},
            {FieldName: r.FieldName, DisplayLabel: r.DisplayLabel,
             Problem: If(
                IsBlank(v), "",
                !(r.FieldName in NumberFields) && (IsBlank(k) || k in Placeholders),
                    "looks like a placeholder. Enter the real value, or ask the MDM team if you don't have it",
                c = "Text", If(Len(k) >= 3, "", "is too short"),
                c = "Email", If(IsMatch(v, Match.Email), "", "isn't a valid email address"),
                c = "RegNo", With({u: Upper(k)}, Switch(ddCustomerCountry.Selected.Value,
                    "SG", If(IsMatch(u, "^(?:\d{8}[A-Z]|\d{9}[A-Z]|[TSR]\d{2}[A-Z]{2}\d{4}[A-Z])$"), "",
                        "doesn't look like a Singapore UEN (e.g. 200288778W)"),
                    "MY", If(IsMatch(u, "^(?:\d{12}|\d{14}|\d{6,7}[A-Z])$"), "",
                        "doesn't look like an SSM registration no. (12 digits, e.g. 201901000005)"),
                    "TH", If(IsMatch(u, "^\d{13}$"), "", "should be the 13-digit Thai tax ID"),
                    "ID", If(IsMatch(u, "^\d{15,16}$"), "", "should be the 15 or 16 digit NPWP (e.g. 01.234.567.8-901.000)"),
                    "VN", If(IsMatch(u, "^(?:\d{10}|\d{13})$"), "", "should be the 10-digit tax code (MST), or 13 digits for a branch"),
                    "")),
                c = "HSCode", If(IsMatch(k, "^\d{6,10}$"), "", "should be 6 to 10 digits (e.g. 3822.19.00)"),
                c = "TaxID", If(Len(k) >= 5 && Len(k) <= 20 && IsMatch(k, "\d", MatchOptions.Contains), "",
                    "should be 5 to 20 letters and digits, including at least one digit"),
                c = "LicenceNo", If(Len(k) >= 5 && Len(k) <= 30 && IsMatch(k, "\d", MatchOptions.Contains), "",
                    "should be 5 to 30 letters and digits, including at least one digit"),
                c = "PartNo", If(Len(k) >= 3, "", "is too short for a part no."),
                c = "BankAccount", If(Len(k) >= 6 && Len(k) <= 34 && IsMatch(k, "\d{4}", MatchOptions.Contains), "",
                    "doesn't look like a bank account no. (6 to 34 characters, mostly digits)"),
                c = "Months", If(Value(v) >= 1 && Value(v) <= 120, "", "should be between 1 and 120 months"),
                c = "Year", If(Value(v) >= Year(Today()) - 1 && Value(v) <= Year(Today()) + 2, "",
                    "should be between " & (Year(Today()) - 1) & " and " & (Year(Today()) + 2)),
                "")}
        )
    )
);
FieldProblems = Filter(FieldChecks, !IsBlank(Problem));
EmailInvalid = "ContactEmail" in FieldProblems.FieldName;

// ---- Live duplicate checks. Customers use the same clean-up as sharepoint/LISTS.md
LegalWords = ["pte", "ltd", "sdn", "bhd", "co", "jsc", "pt", "inc", "gmbh", "ag", "sa", "the"];
NameBasic = Concat(
    Filter(Split(Trim(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Lower(txtCustomerName.Text), ".", " "), ",", " "), "(", " "), ")", " "), "-", " "), "/", " "), "'", " "), """", " "), "&", " "), ":", " "), ";", " "), "#", " "), "+", " ")), " ") As w,
           !IsBlank(w.Value) && !(w.Value in LegalWords)),
    Value, " "
);
NameKey = Concat(
    ForAll(Split(NameBasic, " ") As w,
           Switch(w.Value, "hosp", "hospital", "ctr", "centre", "center", "centre", "univ", "university", "med", "medical", "lab", "laboratory", "spec", "specialist", "diag", "diagnostic", "gen", "general", "comm", "community", w.Value)),
    Value, " "
);
KeyWords = Filter(Split(NameKey, " ") As w, Len(w.Value) >= 4);
CustomerRows = ForAll(
    CustomerMaster,
    {RecordID: Title, Name: CustomerName, Norm: NormalizedName, Country: Country, RegNo: RegistrationNo}
);
RegMatch = If(
    !IsBlank(Trim(txtRegistrationNo.Text)),
    LookUp(CustomerRows, Upper(RegNo) = Upper(Trim(txtRegistrationNo.Text)))
);
AliasMatch = If(!IsBlank(NameBasic), LookUp(Aliases, Title = NameBasic || Title = NameKey));
NameMatch = If(
    CountRows(KeyWords) > 0 && !IsBlank(ddCustomerCountry.Selected.Value),
    First(Filter(CustomerRows As c,
                 c.Country = ddCustomerCountry.Selected.Value &&
                 CountRows(Filter(KeyWords As k, !(k.Value in c.Norm))) = 0))
);
// Items: same manufacturer part no. blocks; all main words of the description in any entity's item warns
ItemKeyWords = Filter(Split(Trim(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(Lower(txtItemDescription.Text), ".", " "), ",", " "), "(", " "), ")", " "), "-", " "), "/", " "), "'", " "), """", " "), "&", " "), ":", " "), ";", " "), "#", " "), "+", " ")), " ") As w, Len(w.Value) >= 4);
ItemRows = ForAll(
    ItemMaster,
    {RecordID: Title, Name: Description, Norm: NormalizedName, PartNo: ManufacturerPartNo, Regulated: Regulated}
);
PartMatch = If(
    !IsBlank(Trim(txtManufacturerPartNo.Text)),
    LookUp(ItemRows, Upper(PartNo) = Upper(Trim(txtManufacturerPartNo.Text)))
);
ItemNameMatch = If(
    CountRows(ItemKeyWords) > 0,
    First(Filter(ItemRows As c, CountRows(Filter(ItemKeyWords As k, !(k.Value in c.Norm))) = 0))
);
// Help under "Is this regulated?": what the closest existing item says, and what "Not sure" does
RegulatedHint = With(
    {m: Coalesce(PartMatch, ItemNameMatch)},
    If(IsBlank(m), "",
       "Similar item " & m.RecordID & " (" & m.Name & ") is " & If(m.Regulated, "regulated. ", "not regulated. ")) &
    "Not sure? Choose Not sure and Quality/Regulatory will decide."
);
// Vendors: same tax ID blocks
VendorRows = ForAll(VendorMaster, {RecordID: Title, Name: VendorName, TaxID: TaxID});
VendorTaxMatch = If(
    !IsBlank(Trim(txtTaxID.Text)),
    LookUp(VendorRows, Upper(TaxID) = Upper(Trim(txtTaxID.Text)))
);
// Cost centres: one per entity and department, so a second one blocks
CostCentreMatch = If(
    !IsBlank(SelectedEntity) && !IsBlank(ddDepartment.Selected.Value),
    LookUp(CostCentres, Entity = SelectedEntity && Department = ddDepartment.Selected.Value)
);

DupStatus = Switch(
    SelectedType,
    "Customer", If(!IsBlank(RegMatch) || !IsBlank(AliasMatch), "Blocked", !IsBlank(NameMatch), "Possible match", "Clear"),
    "Item", If(!IsBlank(PartMatch), "Blocked", !IsBlank(ItemNameMatch), "Possible match", "Clear"),
    "Vendor", If(!IsBlank(VendorTaxMatch), "Blocked", "Clear"),
    "Cost Center", If(!IsBlank(CostCentreMatch), "Blocked", "Clear"),
    "Clear"
);
DupRecordID = If(
    DupStatus = "Clear", Blank(),
    Switch(
        SelectedType,
        "Customer", Coalesce(RegMatch.RecordID, AliasMatch.RecordID, NameMatch.RecordID),
        "Item", Coalesce(PartMatch.RecordID, ItemNameMatch.RecordID),
        "Vendor", VendorTaxMatch.RecordID,
        "Cost Center", CostCentreMatch.Title
    )
);
DupMessage = Switch(
    SelectedType,
    "Customer", If(
        !IsBlank(RegMatch),
            "This customer already exists: " & RegMatch.Name & " (" & RegMatch.RecordID & ") has registration no. " & RegMatch.RegNo & ".",
        !IsBlank(AliasMatch),
            "This customer already exists: " & Char(34) & txtCustomerName.Text & Char(34) & " is a known name for " & AliasMatch.CanonicalName & " (" & AliasMatch.RecordID & ").",
        !IsBlank(NameMatch),
            "Possible match: " & NameMatch.Name & " (" & NameMatch.RecordID & ", " & NameMatch.Country & "). If this is a different customer, confirm it with the switch.",
        ""),
    "Item", If(
        !IsBlank(PartMatch),
            "This item already exists: part no. " & PartMatch.PartNo & " is " & PartMatch.Name & " (" & PartMatch.RecordID & ").",
        !IsBlank(ItemNameMatch),
            "Possible match: " & ItemNameMatch.Name & " (" & ItemNameMatch.RecordID & "). If this is a different item, confirm it with the switch.",
        ""),
    "Vendor", If(!IsBlank(VendorTaxMatch),
        "This vendor already exists: tax ID " & VendorTaxMatch.TaxID & " belongs to " & VendorTaxMatch.Name & " (" & VendorTaxMatch.RecordID & ").", ""),
    "Cost Center", If(!IsBlank(CostCentreMatch),
        "A cost centre already exists for " & SelectedEntity & " " & ddDepartment.Selected.Value & ": " & CostCentreMatch.CostCentreName & " (" & CostCentreMatch.Title & ").", ""),

     Stage: Stage, Active: Active}
);
// Like the flow: the first Roles row for the role and entity (or Any); no "@" in its email = nobody set up.
ApproverPreview = ForAll(
    Filter(RuleRows As x,
        x.Active && (x.RequestType = SelectedType || x.RequestType = "Any") &&
        (x.Entity = SelectedEntity || x.Entity = "Any") &&
        (IsBlank(x.ConditionField) ||
         (x.ConditionField = "IsRegulated" && SelectedType = "Item" && ddIsRegulated.Selected.Value in ["Yes", "Not sure"]) ||
         (x.ConditionField = "SuppliesRegulatedGoods" && SelectedType = "Vendor" && tglSuppliesRegulatedGoods.Value))
    ) As r,
    With(
        {p:LookUp(RolesRows, Role = r.Role && (Entity = SelectedEntity || Entity = "Any"))},
        {Stage: r.Stage, Role: r.Role, Person: Coalesce(p.Name, p.Email), Ok: "@" in p.Email}
    )
);
ApproverGap = CountRows(Filter(ApproverPreview, !Ok)) > 0;
ApproverText = If(
    IsBlank(SelectedEntity), "Choose a business entity to see who will approve.",
    CountRows(ApproverPreview) = 0, "No approval steps are set up for this request type.",
    With(
        {s1: Filter(ApproverPreview, Stage < 1.5), s2: Filter(ApproverPreview, Stage >= 1.5),
         gaps: Filter(ApproverPreview, !Ok)},
        Concat(s1, Role & " (" & If(Ok, Person, "nobody set up") & ")", ", ") &
        If(CountRows(s1) > 1, ", all at the same time", "") &
        If(CountRows(s1) > 0 && CountRows(s2) > 0, "; then ", "") &
        Concat(s2, Role & " (" & If(Ok, Person, "nobody set up") & ")", ", ") & "." &
        If(CountRows(gaps) > 0,
            " No " & Concat(gaps, Role, " or ") & " is set up for " & SelectedEntity &
            ", so the MDM team approves in their place.",
            "")
    )
);

// ---- Rules screen (MDM): approval steps that some active business unit has nobody for
RuleGapText = Concat(
    Filter(BusinessUnits, Active = true) As b,
    Concat(
        Filter(
            Distinct(Filter(RuleRows As x, x.Active && (x.Entity = b.Title || x.Entity = "Any")), Role) As r,
            !("@" in LookUp(RolesRows, Role = r.Value && (Entity = b.Title || Entity = "Any")).Email)
        ) As g,
        b.Title & ": nobody set up as " & g.Value & Char(10)
    )
);
